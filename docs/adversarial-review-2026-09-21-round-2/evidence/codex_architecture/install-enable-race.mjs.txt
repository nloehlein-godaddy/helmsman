import assert from 'node:assert/strict';

const major=Number(process.versions.node.split('.')[0]),id='example.install-race';

 let enabled;let triggered=false;const events=[];

 console.log(JSON.stringify({reproduced:true,events,afterRace:replacement,negativeControl:'already-enabled precheck correctly rejects replacement',severityCandidate:'P2: concurrent enable is lost instead of rejecting replacement'},null,2));
}finally{watcher?.close();c1.dispose();c2.dispose();p1.close();p2.close();await rm(root,{recursive:true,force:true});}
