it('demonstrates a pinned PR rerun deleting an unrelated dirty developer worktree', async () => {
  const root = temporary(); const remote = join(root, 'origin.git'); const agents = join(root, 'agents'); const repo = join(agents, 'repo');
  mkdirSync(agents); git(root, 'init', '--bare', remote); git(root, 'clone', remote, repo);
  git(repo, 'config', 'user.name', 'Audit'); git(repo, 'config', 'user.email', 'audit@localhost');
  writeFileSync(join(repo, 'tracked.txt'), 'initial\n'); git(repo, 'add', 'tracked.txt'); git(repo, 'commit', '-m', 'initial');
  git(repo, 'branch', 'feature'); git(repo, 'push', 'origin', 'feature'); const expectedHead = git(repo, 'rev-parse', 'feature');
  const developerWorktree = join(root, 'developer-work-in-progress'); git(repo, 'worktree', 'add', developerWorktree, 'feature');
  writeFileSync(join(developerWorktree, 'tracked.txt'), 'uncommitted valuable edit\n'); writeFileSync(join(developerWorktree, 'untracked.txt'), 'untracked valuable file\n');
  await expect(createWorktreeFromBranch(agents, 'owner/repo', 'stale-pr-rerun', 'feature', 'b'.repeat(40))).rejects.toThrow('PR branch changed');
  expect(readFileSync(join(developerWorktree, 'untracked.txt'), 'utf8')).toBe('untracked valuable file\n');
  const launched = await createWorktreeFromBranch(agents, 'owner/repo', 'new-pr-rerun', 'feature', expectedHead);
  writeFileSync(join(probeRoot, 'dirty-worktree-rerun.json'), JSON.stringify({ probe: 'dirty-worktree-rerun', developerWorktreeExists: existsSync(developerWorktree), untrackedFileExists: existsSync(join(developerWorktree, 'untracked.txt')), newWorktreeExists: existsSync(launched.path) }, null, 2));
  expect(existsSync(developerWorktree)).toBe(false); expect(existsSync(launched.path)).toBe(true);
});

it('demonstrates permanent capacity retention after a definitely pre-launch checkout failure', async () => {
  const root = temporary(); const path = join(root, 'state.sqlite'); let persistence = openPersistence(path);
  cleanups.push(() => persistence.close());
  let db = openDb(persistence.sql); let executions = openExecutionStore(persistence); let admissions = openAdmissionStore(persistence, executions);
  let durable = createRunnerDurability({ persistence, db, executions, admissions, runsDir: root, maxConcurrency: 1, snapshot, verify: async () => {} });
  admissions.admit({ runId: 'failed-preparation', repo: task.repo, request: {}, settings: {}, artifacts: [artifact] }, 1);
  const launch = vi.fn(async () => { throw new Error('worker must never launch'); });
  const createWorktree = vi.fn(async () => { throw new Error('temporary checkout unavailable'); });
  let deps: RunnerDeps = { db, durable, bus: new RunBus(), runsDir: root, genId: () => 'failed-preparation', now: () => new Date().toISOString(), adapter: { id: 'audit', buildCommand: () => ({ cmd: 'fake-worker', args: [] }), parseLine: () => null }, host: { kind: 'detached', launch, stop: vi.fn(async () => {}), isAlive: async () => false }, createWorktree, removeWorktree: async () => {} };
  await startRun(task, deps); expect(launch).not.toHaveBeenCalled(); expect(admissions.get('failed-preparation')?.state).toBe('started');
  persistence.close(); persistence = openPersistence(path); db = openDb(persistence.sql); executions = openExecutionStore(persistence); admissions = openAdmissionStore(persistence, executions);
  durable = createRunnerDurability({ persistence, db, executions, admissions, runsDir: root, maxConcurrency: 1, snapshot, verify: async () => {} });
  createWorktree.mockImplementation(async () => ({ path: root, branch: 'fixed' }));
  deps = { ...deps, db, durable };
  const row = db.getRun('failed-preparation')!; const recoveries = [];
  for (let i = 0; i < 3; i++) recoveries.push(await reattachRun(row, deps));
  writeFileSync(join(probeRoot, 'preparation-failure-restart.json'), JSON.stringify({ probe: 'preparation-failure-restart', admissionState: admissions.get(row.id)?.state, runState: db.getRun(row.id)?.status, preparation: executions.attempt({ runId: row.id, attempt: 1 })?.preparation, recoveries, reservations: executions.reservations(), checkoutCalls: createWorktree.mock.calls.length, launchCalls: launch.mock.calls.length }, null, 2));
  expect(recoveries).toEqual(Array.from({ length: 3 }, () => ({ status: 'blocked', reason: 'Prepared worktree requires reconciliation before attachment' })));
  expect(() => admissions.fail(row.id, 'verified never launched')).toThrow('Attempted execution cannot be abandoned');
  expect(() => executions.releaseReservation(row.id, { neverLaunched: true })).toThrow('Execution ownership is unresolved');
  expect(() => admissions.admit({ runId: 'fresh-independent-run', repo: 'owner/other', request: {}, settings: {}, artifacts: [artifact] }, 1)).toThrow('capacity');
  expect(createWorktree).toHaveBeenCalledTimes(1); expect(launch).not.toHaveBeenCalled();
});

it('demonstrates wrapper stop recording terminal exit while an ordinary descendant keeps writing', async () => {
  const root = temporary(); const worker = join(root, 'worker.mjs'); const descendant = join(root, 'descendant.mjs');
  const ready = join(root, 'descendant-ready'); const afterStop = join(root, 'write-after-worker-exit');
  const exited = join(root, 'descendant-exited'); const specPath = join(root, 'spec.json');
  writeFileSync(descendant, `import { writeFileSync } from 'node:fs';\nwriteFileSync(${JSON.stringify(ready)}, 'ready');\nsetTimeout(() => { writeFileSync(${JSON.stringify(afterStop)}, 'mutation after stop'); writeFileSync(${JSON.stringify(exited)}, 'done'); }, 750);\n`);
  writeFileSync(worker, `import { spawn } from 'node:child_process';\nspawn(process.execPath, [${JSON.stringify(descendant)}], { stdio: 'ignore' });\nsetTimeout(() => {}, 4000);\n`);
  const spec = { version: 1, runId: 'stop-audit', launchId: 'stop-audit-launch', attempt: 1, cmd: process.execPath, args: [worker], cwd: root, logPath: join(root, 'worker.log'), exitPath: join(root, 'worker.exit'), receiptPath: join(root, 'worker.receipt') };
  writeFileSync(specPath, JSON.stringify(spec));
  const wrapper = spawn(process.execPath, [join(import.meta.dirname, 'run-wrapper.mjs'), specPath], { detached: true, stdio: 'ignore' });
  cleanups.push(() => { if (wrapper.exitCode === null && wrapper.signalCode === null) wrapper.kill('SIGTERM'); });
  const completion = once(wrapper, 'exit');
  await vi.waitFor(() => expect(existsSync(ready)).toBe(true));
  const ownership = JSON.parse(readFileSync(spec.receiptPath, 'utf8'));
  expect(ownership.pid).toBe(wrapper.pid);
  await createDetachedExecutionHost().stop({ version: 1, identity: { runId: spec.runId, launchId: spec.launchId, attempt: spec.attempt, receiptPath: spec.receiptPath }, state: { kind: 'detached', pid: ownership.pid, processStart: ownership.processStart } });
  await completion;
  const receipt = JSON.parse(readFileSync(spec.exitPath, 'utf8'));
  const mutationAtTerminalReceipt = existsSync(afterStop);
  await vi.waitFor(() => expect(existsSync(exited)).toBe(true), { timeout: 2000 });
  const mutationAfterStop = readFileSync(afterStop, 'utf8');
  writeFileSync(join(probeRoot, 'descendant-after-stop.json'), JSON.stringify({ probe: 'descendant-after-stop', receipt, mutationAtTerminalReceipt, mutationAfterStop }, null, 2));
  expect(receipt.exitCode).toBe(1); expect(mutationAtTerminalReceipt).toBe(false); expect(mutationAfterStop).toBe('mutation after stop');
});

  const loaded = await loadRetainedAgentProvider({ pluginsRoot: root, runtimeArtifactsRoot: root, record, artifacts: [{ pluginId: 'audit.parser', artifactHash: archive.hash, packageVersion: '1.0.0', requires: [] }], signal: controller.signal });

  const durable = createRunnerDurability({ persistence, db, executions, runsDir: root, maxConcurrency: 1, snapshot, verify: async () => {} });
  let stopped = false; let savedSpec: any; let stopTimer: ReturnType<typeof setTimeout> | undefined;
  const ref = { kind: 'detached' as const, pid: 123 };
  const stop = vi.fn(async () => { const receipt = JSON.parse(readFileSync(savedSpec.receiptPath, 'utf8')); writeFileSync(savedSpec.logPath, 'worker stopped cleanly\n'); writeFileSync(savedSpec.exitPath, JSON.stringify({ ...receipt, exitCode: 1 })); });
