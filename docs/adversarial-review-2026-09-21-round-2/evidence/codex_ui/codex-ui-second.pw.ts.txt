test('installed shell routes, keyboard navigation, malformed data, and responsive presentation', async ({ page }) => {
  test.setTimeout(180_000);
  const host = await fixture({ distribution });
  host.env.REPO_PROJECT_MAP = 'org/a=TASK,org/b=OTHER';
  const pageErrors: string[] = [];
  const consoleErrors: string[] = [];
  page.on('pageerror', error => pageErrors.push(error.message));
  page.on('console', message => { if (message.type() === 'error') consoleErrors.push(message.text()); });
  try {
    const origin = await host.start();
    await page.goto(`${origin}/helm?repo=org/a`);
    await ready(page);
    const notifications = page.getByRole('button', { name: /^Notifications,/ });
    await notifications.focus();
    await notifications.click();
    await expect(notifications).toHaveAttribute('aria-expanded', 'true');
    await page.keyboard.press('Escape');
    await expect(notifications).toHaveAttribute('aria-expanded', 'false');
    await expect(notifications).toBeFocused();
    const tabs = page.locator('.page-tab');
    await tabs.filter({ hasText: 'Helm' }).focus();
    await page.keyboard.press('ArrowRight');
    await expect(tabs.nth(1)).toBeFocused();
    await page.keyboard.press('Enter');

    const inventory: unknown[] = [];
    for (const route of routes) {
      await page.goto(`${origin}${route}?repo=org/a`);
      await ready(page);
      inventory.push({ route, headings: await page.getByRole('heading').allTextContents(), text: (await page.locator('#page-content').innerText()).slice(0, 300), overflow: await page.evaluate(() => document.documentElement.scrollWidth > innerWidth + 1) });
    }
    await writeFile(join(artifacts, 'route-inventory.json'), JSON.stringify(inventory, null, 2));
    expect(inventory.some((value: any) => value.overflow)).toBe(false);

    await page.route('**/api/run-inspect/history?**', route => route.fulfill({ status: 200, contentType: 'application/json', body: 'null' }));
    await page.goto(`${origin}/run-history?repo=org/a`);
    await ready(page);
    await expect(page.locator('section.plugin-run-history')).toContainText('Run history is unavailable. Refresh to retry.');
    await page.unroute('**/api/run-inspect/history?**');

    await page.route('**/api/run-inspect/history?**', route => route.fulfill({ status: 200, contentType: 'application/json', body: JSON.stringify({ status: 'available', value: { repo: 'org/a', items: [null] } }) }));
    await page.reload();
    await ready(page);
    await expect(page.locator('section.plugin-run-history')).toContainText('Run history is unavailable. Refresh to retry.');
    await page.unroute('**/api/run-inspect/history?**');

    await page.setViewportSize({ width: 390, height: 844 });
    for (const route of ['/helm', '/todos', '/runs', '/outcomes', '/communication', '/local-git', '/config']) {
      await page.goto(`${origin}${route}?repo=org/a`);
      await ready(page);
      expect(await page.evaluate(() => document.documentElement.scrollWidth <= innerWidth + 1), route).toBe(true);
    }
    await page.goto(`${origin}/recovery?repo=org/a`);
    await expect(page.getByRole('heading', { name: 'Recovery', exact: true })).toBeVisible();
    expect(await page.evaluate(() => document.documentElement.scrollWidth <= innerWidth + 1)).toBe(true);
    await page.screenshot({ path: join(artifacts, 'recovery-mobile.png'), fullPage: true });

    expect(pageErrors).toEqual([]);
    expect(consoleErrors.filter(value => !value.includes('Failed to load resource'))).toEqual([]);
  } finally { await host.close(); }
});

test('request guard accepts installed clients and rejects cross-origin or non-JSON mutations before parsing', async ({ page, request }) => {
  test.setTimeout(90_000);
  const host = await fixture({ distribution });
  host.env.REPO_PROJECT_MAP = 'org/a=TASK';
  try {
    const origin = await host.start();
    await page.goto(`${origin}/helm?repo=org/a`);
    await ready(page);
    const results = await page.evaluate(async () => {
      const calls = [
        ['/api/repos/org%2Fa/auto-claim', { method: 'POST', headers: { 'Content-Type': 'application/json' }, body: JSON.stringify({ enabled: true }) }],
        ['/api/agents/missing/stop', { method: 'POST', headers: { 'Content-Type': 'application/json' } }],
        ['/api/config', { method: 'PUT', headers: { 'Content-Type': 'application/json' }, body: JSON.stringify({ key: 'PIRATE_MODE', value: 'false' }) }],
      ] as const;
      return Promise.all(calls.map(async ([path, init]) => { const response = await fetch(path, init); return { path, status: response.status, body: await response.text() }; }));
    });
    expect(results.slice(0, 2).map(value => value.status)).toEqual([200, 200]);
    expect(results[2]?.status).not.toBe(403);
    const textPlain = await request.post(`${origin}/api/repos/org%2Fa/auto-claim`, { headers: { 'Content-Type': 'text/plain' }, data: JSON.stringify({ enabled: false }) });
    expect(textPlain.status()).toBe(403);
    const crossOrigin = await request.post(`${origin}/api/repos/org%2Fa/auto-claim`, { headers: { Origin: 'http://127.0.0.1:1', 'Content-Type': 'application/json' }, data: { enabled: false } });
    expect(crossOrigin.status()).toBe(403);
    const malformed = await page.evaluate(async path => { const response = await fetch(path, { method: 'POST', headers: { 'Content-Type': 'application/json' }, body: '{' }); return { status: response.status, body: await response.text() }; }, '/api/repos/org%2Fa/auto-claim');
    const stateAfterMalformed = await (await request.get(`${origin}/api/agents`)).json();
    expect(malformed.status).toBe(200);
    expect(malformed.body).toContain('"enabled":false');
    expect(stateAfterMalformed.autoClaim).toEqual([]);
    const malformedControl = await page.evaluate(async path => { const response = await fetch(path, { method: 'PUT', headers: { 'Content-Type': 'application/json' }, body: '{' }); return { status: response.status, body: await response.text() }; }, '/api/config');
    expect(malformedControl.status).toBe(400);
    await writeFile(join(artifacts, 'request-guard-browser-clients.json'), JSON.stringify({ results, textPlain: textPlain.status(), crossOrigin: crossOrigin.status(), malformed, stateAfterMalformed, malformedControl }, null, 2));
  } finally { await host.close(); }
});

  const host = await fixture({ distribution });
  host.env.REPO_PROJECT_MAP = 'org/a=TASK,org/b=OTHER';
  const pageErrors: string[] = [];
  page.on('pageerror', error => pageErrors.push(error.message));

    await page.route('**/api/source-commands', async route => {
      submitted = route.request().postDataJSON();

      expect(response.ok()).toBe(true);
      await route.abort('failed');
    }, { times: 1 });
    await page.getByRole('button', { name: 'Create Todo', exact: true }).click();
    await expect(page.getByRole('button', { name: 'Check request', exact: true })).toBeVisible();
