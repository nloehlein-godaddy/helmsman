import assert from 'node:assert/strict';
import { cp, mkdtemp, mkdir, readFile, symlink, writeFile } from 'node:fs/promises';
import { join } from 'node:path';
import { spawnSync } from 'node:child_process';
import { fileURLToPath } from 'node:url';
import { pluginCommand } from '../../integration/release/server/helmsman/plugins/cli.js';
import { loadPluginSnapshot } from '../../integration/release/server/helmsman/plugins/store.js';

const directory = await mkdtemp(new URL('./author-', import.meta.url).pathname);
const author = join(directory, 'project');
const release = fileURLToPath(new URL('../../integration/release', import.meta.url));
await cp(join(release, 'examples/plugins/scope-panel'), author, { recursive: true });
await mkdir(join(author, 'node_modules/@helmsman'), { recursive: true });
await symlink(join(release, 'sdk'), join(author, 'node_modules/@helmsman/plugin-sdk'));
const compiler = fileURLToPath(new URL('../../codex_architecture/node_modules/typescript/bin/tsc', import.meta.url));
const built = spawnSync(process.execPath, [compiler, '-p', author, '--listFiles'], { cwd: author, encoding: 'utf8' });
await writeFile(join(directory, 'compiler-files.log'), built.stdout + built.stderr);
assert.equal(built.status, 0, built.stdout + built.stderr);
assert.ok(!built.stdout.includes('/codex_architecture/src/'));
await cp(join(author, 'plugin-package.json'), join(author, 'dist/plugin-package.json'));
const archivePath = join(directory, 'author.hmp');
const root = join(directory, 'plugins'); const dbPath = join(directory, 'main.sqlite');
for (const args of [ ['pack', join(author, 'dist'), archivePath], ['install', archivePath], ['enable', 'example.scope-panel'] ]) {
  console.log(await pluginCommand(['--dir', root, '--db', dbPath, ...args]));
}
const snapshot = await loadPluginSnapshot(root);
assert.equal(snapshot.errors.length, 0);
assert.equal(snapshot.catalog.plugins[0]?.package.manifest.manifestVersion, 2);
assert.equal(snapshot.catalog.plugins[0]?.package.manifest.panels.length, 1);
assert.equal(snapshot.catalog.plugins[0]?.package.manifest.pages.length, 1);
console.log('PASS: release-only public SDK frontend compiles, packs, installs and enables without host source imports; fixture', directory);
