import assert from 'node:assert/strict';
import { createServer as createHttpServer } from 'node:http';
import { createServer as createViteServer } from 'vite';
import { chromium } from '@playwright/test';
import { allowMutationRequest } from './server/helmsman/request-origin';

const observed: Array<{ host?: string; origin?: string; allowed: boolean }> = [];
let mutations = 0;
const backend = createHttpServer((req, res) => {
  if (req.method === 'GET') { res.writeHead(200, { 'Content-Type': 'text/html' }); res.end('<!doctype html><title>Disposable dev proxy fixture</title>'); return; }
  const allowed = allowMutationRequest(req, new URL(req.url ?? '/', 'http://localhost').pathname);
  observed.push({ host: req.headers.host, origin: req.headers.origin, allowed });
  if (allowed) mutations++;
  res.writeHead(allowed ? 200 : 403, { 'Content-Type': 'application/json' }); res.end(JSON.stringify({ mutations }));
});
const foreign = createHttpServer((_req, res) => { res.writeHead(200, { 'Content-Type': 'text/html' }); res.end('<!doctype html><title>Foreign fixture</title>'); });
let vite: Awaited<ReturnType<typeof createViteServer>> | undefined;
let browser: Awaited<ReturnType<typeof chromium.launch>> | undefined;
try {
  await new Promise<void>(resolve => backend.listen(0, 'localhost', resolve));
  const backendAddress = backend.address(); assert(backendAddress && typeof backendAddress !== 'string');
  process.env.HELMSMAN_PORT = String(backendAddress.port);
  vite = await createViteServer({ configFile: './vite.config.ts', server: { host: '127.0.0.1', port: 0, hmr: false, watch: null }, logLevel: 'error' });
  await vite.listen();
  const viteAddress = vite.httpServer?.address(); assert(viteAddress && typeof viteAddress !== 'string');
  const origin = `http://127.0.0.1:${viteAddress.port}`;
  browser = await chromium.launch(); const page = await browser.newPage();
  await page.goto(`${origin}/api/ping`);
  const same = await page.evaluate(async () => {
    const response = await fetch('/api/fixture/mutate', { method: 'POST', headers: { 'Content-Type': 'application/json' }, body: '{}' });
    return { status: response.status, body: await response.json() };
  });
  assert.deepEqual(same, { status: 200, body: { mutations: 1 } });
  assert.deepEqual(observed[0], { host: new URL(origin).host, origin, allowed: true });
  await new Promise<void>(resolve => foreign.listen(0, '127.0.0.1', resolve));
  const foreignAddress = foreign.address(); assert(foreignAddress && typeof foreignAddress !== 'string');
  await page.goto(`http://127.0.0.1:${foreignAddress.port}`);
  const rejected = page.waitForResponse(response => response.url() === `${origin}/api/fixture/mutate`);
  const opaque = await page.evaluate(async origin => {
    const response = await fetch(`${origin}/api/fixture/mutate`, { method: 'POST', mode: 'no-cors', headers: { 'Content-Type': 'text/plain' }, body: '{}' });
    return { type: response.type, status: response.status };
  }, origin);
  assert.deepEqual(opaque, { type: 'opaque', status: 0 });
  assert.equal((await rejected).status(), 403); assert.equal(mutations, 1);
  console.log(JSON.stringify({ sameOrigin: same, forwardedHeaders: observed, foreignResponse: opaque, mutations }, null, 2));
} finally {
  await browser?.close(); await vite?.close();
  await Promise.all([new Promise<void>(resolve => backend.close(() => resolve())), new Promise<void>(resolve => foreign.close(() => resolve()))]);
}
